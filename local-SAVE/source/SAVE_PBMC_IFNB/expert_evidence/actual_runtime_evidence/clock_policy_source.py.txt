def signature(path):
 tree=ast.parse(path.read_text())
 for node in ast.walk(tree):
  if isinstance(node,(ast.Module,ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef)) and node.body and isinstance(node.body[0],ast.Expr) and isinstance(node.body[0].value,ast.Constant) and isinstance(node.body[0].value.value,str):node.body.pop(0)
 return hashlib.sha256(ast.dump(tree,include_attributes=False).encode()).hexdigest()
def union(values):

def witnessed_pending_queue_intervals(relay):
 # An unfinished numerical request has no final queue_seconds_excluded yet.
 # Derive only its own actually observed BUSY admission interval from local
 # transport receipt epochs. Never infer a future wait from uptime, a PID,
 # remote wall timestamps or a still DISPATCHED worker that might be running.
 intervals=[]
 for folder in relay.glob('evaluation-*'):
  if (folder/'result.json').exists():continue
  observed=[]
  for log in folder.glob('*-transport.original.jsonl'):
   for line in log.read_text().splitlines():
    try:
     row=json.loads(line);response=json.loads(row['stdout'].splitlines()[-1])
    except (ValueError,KeyError,IndexError):continue
    epoch=row.get('epoch');status=response.get('status')
    if isinstance(epoch,(int,float)) and row.get('exit_code')==0:
     observed.append((epoch,row.get('operation'),status))
  if any(status in ('RUNNING','COMPLETE') for _,_,status in observed):continue
  dispatched=[epoch for epoch,operation,status in observed if operation=='submit' and status=='DISPATCHED']
  busy=[epoch for epoch,_,status in observed if status=='BUSY']
  if dispatched and busy:
   intervals.append((min(dispatched),max(busy)))
 return union(intervals)
